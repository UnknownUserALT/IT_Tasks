-- 1. Создание таблиц
CREATE TABLE students (
    student_id SERIAL PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    group_number VARCHAR(20) NOT NULL
);

CREATE TABLE subjects (
    subject_id SERIAL PRIMARY KEY,
    subject_name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE grades (
    grade_id SERIAL PRIMARY KEY,
    student_id INT REFERENCES students(student_id) ON DELETE CASCADE,
    subject_id INT REFERENCES subjects(subject_id) ON DELETE CASCADE,
    grade INT CHECK (grade >= 1 AND grade <= 5)
);

CREATE TABLE attendance (
    attendance_id SERIAL PRIMARY KEY,
    student_id INT REFERENCES students(student_id) ON DELETE CASCADE,
    date_attended DATE NOT NULL,
    status VARCHAR(20) CHECK (status IN ('присутствовал', 'отсутствовал', 'болел'))
);

CREATE TABLE notes (
    note_id SERIAL PRIMARY KEY,
    student_id INT REFERENCES students(student_id) ON DELETE CASCADE,
    note_text TEXT NOT NULL
);

-- 4. Создание представлений

-- Представление для вычисления среднего балла каждого студента по всем предметам
CREATE VIEW student_avg_grades AS
SELECT 
    s.student_id,
    s.full_name,
    s.group_number,
    ROUND(AVG(g.grade)::numeric, 2) as average_grade,
    COUNT(g.grade_id) as total_grades
FROM students s
LEFT JOIN grades g ON s.student_id = g.student_id
GROUP BY s.student_id, s.full_name, s.group_number
ORDER BY average_grade DESC NULLS LAST;

-- 6. SQL-запросы

-- 1. Найти 5 одногруппников студента (2 до и 3 после него по списку student_id)
WITH target_student AS (
    SELECT student_id, group_number
    FROM students 
    WHERE full_name = 'Сидорова Анна Михайловна'
)
SELECT s.*
FROM students s, target_student ts
WHERE s.group_number = ts.group_number 
  AND s.student_id BETWEEN ts.student_id - 2 AND ts.student_id + 3
  AND s.student_id != ts.student_id
ORDER BY s.student_id
LIMIT 5;

-- 2. Посмотреть средний балл конкретного студента через представление
SELECT *
FROM student_avg_grades
WHERE full_name = 'Иванов Иван Иванович';

-- 3. Агрегировать средний балл по предмету «Информатика»
SELECT 
    s.subject_name,
    ROUND(AVG(g.grade)::numeric, 2) as average_grade,
    COUNT(g.grade_id) as total_grades,
    COUNT(DISTINCT g.student_id) as students_count
FROM subjects s
JOIN grades g ON s.subject_id = g.subject_id
WHERE s.subject_name = 'Информатика'
GROUP BY s.subject_id, s.subject_name;

-- 4. Полнотекстовый поиск по заметкам, найти заметки, содержащие слово «Информатика»

SELECT 
    n.note_id,
    s.full_name,
    s.group_number,
    n.note_text
FROM notes n
JOIN students s ON n.student_id = s.student_id
WHERE n.note_text ~* 'информатик';

-- 5. Обновить посещаемость студента на конкретную дату через транзакцию
