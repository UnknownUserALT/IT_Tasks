1) Создание базы данных:

CREATE DATABASE task_management;


2) Создание таблиц:

CREATE TABLE users (
    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE tasks (
    id INT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    priority VARCHAR(20) CHECK (priority IN ('Высокий', 'Средний', 'Низкий')) NOT NULL,
    due_date DATE NOT NULL,
    status VARCHAR(20) CHECK (status IN ('Выполнена', 'В процессе', 'Отложена')) NOT NULL
);

CREATE TABLE task_assignments (
    assignment_id INT PRIMARY KEY,
    task_id INT NOT NULL,
    user_id INT NOT NULL
);

3) Связи между таблицами:

ALTER TABLE task_assignments 
ADD CONSTRAINT fk_task_assignments_tasks 
FOREIGN KEY (task_id) REFERENCES tasks(id) 
ON DELETE CASCADE;

ALTER TABLE task_assignments 
ADD CONSTRAINT fk_task_assignments_users 
FOREIGN KEY (user_id) REFERENCES users(id) 
ON DELETE CASCADE;

5) Запросы:


SELECT id, title, priority, due_date, status 
FROM tasks 
ORDER BY due_date;

SELECT t.id, t.title, t.priority, t.due_date, t.status 
FROM tasks t
JOIN task_assignments ta ON t.id = ta.task_id
WHERE ta.user_id = 1;


SELECT id, title, priority, due_date, status 
FROM tasks 
WHERE priority = 'Высокий' 
AND due_date BETWEEN '2025-11-21' AND '2025-11-28'
AND status != 'Выполнена';


SELECT status, COUNT(*) as task_count 
FROM tasks 
GROUP BY status;

6) Доп. Задание:

-- Отчет о просроченных задачах и ответственных пользователях
SELECT 
    t.id,
    t.title,
    t.priority,
    t.due_date,
    t.status,
    u.name as responsible_user
FROM tasks t
JOIN task_assignments ta ON t.id = ta.task_id
JOIN users u ON ta.user_id = u.id
WHERE t.due_date < '2025-11-21' 
AND t.status != 'Выполнена'
ORDER BY 
    CASE 
        WHEN t.priority = 'Высокий' THEN 1
        WHEN t.priority = 'Средний' THEN 2
        WHEN t.priority = 'Низкий' THEN 3
    END,
    t.due_date;
